**Goal:** Put Aaron's GitHub contribution skyline inside "What I'm up to right now". It shows six months of real data that refreshes daily, sits beside its stats, stays flat until it is seen, morphs once, and has a Flat/Skyline toggle. When the token is missing or GitHub fails, the block falls back to a committed snapshot with an "As of" stamp, so it is never blank.

**Architecture:**
- **Data.** `lib/metrics/github.ts` is server-only. It fetches the calendar through GraphQL with `GITHUB_CONTRIB_TOKEN`, validates the response with zod, computes the window with the pure `lib/metrics/window.ts`, and returns `{ data, error }`. `loadContributions()` falls back to the last good fetch (in memory, marked stale) or to `lib/metrics/snapshot.ts` (also marked stale).
- **Server component.** `components/metrics/Metrics.tsx` is an async Server Component. It renders `MetricsStats` (server) next to `MetricsChart` (client). The chart is the lab's skyline engine, lifted with the pure maths in `lib/metrics/skyline/maths.ts` and the canvas code in `components/metrics/skyline/*`, and driven by `useScrollMorph` (one ScrollTrigger plus the pure `lib/metrics/seenGate.ts`).
- **Revalidation.** `app/page.tsx` sets `export const revalidate = 86400`. `next.config.mjs` has no `cacheComponents`, so `"use cache"` and `cacheLife` are off. Turning them on would move the whole app to Cache Components, where every uncached read needs Suspense or a cache, which is a site-wide change and not part of this slice. `unstable_cache` is the superseded API in 16 and adds nothing over fetch's own `next.revalidate`. A route-segment `revalidate` is already proven on this exact version by `app/recruiting/page.tsx` (`revalidate = 900`).

**Review Focus: five failure modes the tests do not cover**

`lib/metrics/window.ts`:
```ts
import type { ContributionDay, ContributionWindow, Contributions, Streak } from "./types";

// The window and the figures, pure. Dates are UTC calendar days
// ("YYYY-MM-DD"); scripts/fetch-contributions.mjs mirrors windowStart.

export const DAY_MS = 86_400_000;
export const isoDay = (ms: number) => new Date(ms).toISOString().slice(0, 10);
const utc = (date: string) => Date.UTC(+date.slice(0, 4), +date.slice(5, 7) - 1, +date.slice(8, 10));

// 6mo: six months back, then to that week's Sunday, so the grid's first
// column is a whole week. 12mo: GitHub's rolling year. year: January 1.
export function windowStart(today: string, window: ContributionWindow): string {
  const y = +today.slice(0, 4);
  const m = +today.slice(5, 7) - 1;
  const d = +today.slice(8, 10);
  if (window === "year") return `${y}-01-01`;
  if (window === "12mo") return isoDay(Date.UTC(y - 1, m, d + 1));
  const back = Date.UTC(y, m - 6, d);
  return isoDay(back - new Date(back).getUTCDay() * DAY_MS);
}

export function fillDays(days: ContributionDay[], from: string, to: string): ContributionDay[] {
  const counts = new Map(days.map((x) => [x.date, x.count]));
  const out: ContributionDay[] = [];
  for (let ms = utc(from), end = utc(to); ms <= end; ms += DAY_MS) {
    const date = isoDay(ms);
    out.push({ date, count: counts.get(date) ?? 0 });
  }
  return out;
}

// The run that ends today, or yesterday when today has nothing yet (today is not over).
export function currentStreak(days: ContributionDay[]): Streak {
  let end = days.length - 1;
  if (end >= 0 && days[end].count === 0) end -= 1;
  let start = end;
  while (start >= 0 && days[start].count > 0) start -= 1;
  const n = end - start;
  return n > 0 ? { days: n, start: days[start + 1].date, end: days[end].date } : { days: 0, start: null, end: null };
}

export function summarize(days: ContributionDay[], from: string, to: string): Omit<Contributions, "window" | "stale"> {
  const filled = fillDays(days, from, to);
  return {
    range: { from, to },
    days: filled,
    total: filled.reduce((sum, x) => sum + x.count, 0),
    activeDays: filled.filter((x) => x.count > 0).length,
    streak: currentStreak(filled),
  };
}
```

```ts
import "server-only";
import { z } from "zod";
import { snapshotContributions } from "./snapshot";
import { isoDay, summarize, windowStart } from "./window";
import type { ContributionWindow, Contributions, Result } from "./types";

// The contribution calendar through GitHub's GraphQL API with a token that
// has no repository scopes (the profile setting "Private contributions" puts
// private counts in the calendar). The home route revalidates daily
// (app/page.tsx); this fetch carries the same window. On failure the page
// gets the last good fetch this instance saw, else the committed snapshot,
// both stamped stale: never blank. The /recruiting pattern
// (lib/recruiting/data.ts).

export const GITHUB_LOGIN = "aaron-sulbaran";
export const REVALIDATE_SECONDS = 86_400;

const QUERY = `query($login: String!, $from: DateTime!, $to: DateTime!) {
  user(login: $login) { contributionsCollection(from: $from, to: $to) {
    contributionCalendar { weeks { contributionDays { date contributionCount } } } } } }`;

const Calendar = z.object({
  data: z
    .object({
      user: z
        .object({
          contributionsCollection: z.object({
            contributionCalendar: z.object({
              weeks: z.array(
                z.object({
                  contributionDays: z.array(
                    z.object({ date: z.string().regex(/^\d{4}-\d{2}-\d{2}$/), contributionCount: z.number().int().min(0) }),
                  ),
                }),
              ),
            }),
          }),
        })
        .nullable(),
    })
    .nullable(),
  errors: z.array(z.object({ message: z.string() })).optional(),
});

function explain(status: number): string {
  if (status === 401) return "GitHub rejected GITHUB_CONTRIB_TOKEN (expired, revoked, or mistyped).";
  if (status === 403 || status === 429) return `GitHub refused the request (rate limit), status ${status}.`;
  return `GitHub's GraphQL API answered ${status}.`;
}

export async function fetchContributions(token: string, today: string, window: ContributionWindow = "6mo"): Promise<Result<Contributions>> {
  const from = windowStart(today, window);
  try {
    const res = await fetch("https://api.github.com/graphql", {
      method: "POST",
      headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" },
      body: JSON.stringify({ query: QUERY, variables: { login: GITHUB_LOGIN, from: `${from}T00:00:00Z`, to: `${today}T23:59:59Z` } }),
      next: { revalidate: REVALIDATE_SECONDS },
      signal: AbortSignal.timeout(8000),
    });
    if (!res.ok) return { data: null, error: explain(res.status) };
    const parsed = Calendar.safeParse(await res.json());
    if (!parsed.success) {
      const issues = parsed.error.issues.map((i) => `${i.path.join(".")} ${i.message}`).join("; ");
      return { data: null, error: `GitHub's calendar did not match the expected shape: ${issues}` };
    }
    const user = parsed.data.data?.user;
    if (!user) {
      const messages = parsed.data.errors?.map((e) => e.message).join("; ");
      return { data: null, error: messages || `GitHub returned no user for ${GITHUB_LOGIN}.` };
    }
    const days = user.contributionsCollection.contributionCalendar.weeks
      .flatMap((w) => w.contributionDays)
      .map((d) => ({ date: d.date, count: d.contributionCount }));
    return { data: { window, ...summarize(days, from, today), stale: false }, error: null };
  } catch (error) {
    return { data: null, error: error instanceof Error ? error.message : String(error) };
  }
}

let lastGood: Contributions | null = null;

export async function loadContributions(now: Date = new Date()): Promise<Contributions> {
  const token = process.env.GITHUB_CONTRIB_TOKEN;
  if (!token) return snapshotContributions();
  const { data, error } = await fetchContributions(token, isoDay(now.getTime()));
  if (data) {
    lastGood = data;
    return data;
  }
  console.error("[metrics] contribution fetch failed:", error);
  return lastGood ? { ...lastGood, stale: true } : snapshotContributions();
}
```

```ts
import type { MetricsCopy } from "@/lib/content";
import type { Contributions } from "./types";

// The stats beside the skyline, as strings. Dates read "Aug 8", never a dashed range.
const numberFormat = new Intl.NumberFormat("en-US");
const monthDay = new Intl.DateTimeFormat("en-US", { month: "short", day: "numeric", timeZone: "UTC" });

export const formatCount = (n: number) => numberFormat.format(n);
export const shortDate = (date: string) => monthDay.format(new Date(`${date}T00:00:00Z`));

export type MetricsStat = { key: string; value: string; unit: string | null; label: string; sub: string | null };

// The current streak leads (the one figure that describes now); the longest
// streak is never rendered. Slots appear only when content.ts fills them.
export function metricsStats(c: Contributions, copy: MetricsCopy): MetricsStat[] {
  const unit = (n: number) => (n === 1 ? copy.dayUnit.one : copy.dayUnit.other);
  const stats: MetricsStat[] = [
    {
      key: "streak",
      value: formatCount(c.streak.days),
      unit: unit(c.streak.days),
      label: copy.streakLabel,
      sub: c.streak.start ? `${copy.since} ${shortDate(c.streak.start)}` : null,
    },
    { key: "total", value: formatCount(c.total), unit: null, label: copy.totalLabel, sub: null },
    { key: "active", value: formatCount(c.activeDays), unit: null, label: copy.activeLabel, sub: null },
  ];
  for (const [key, slot] of Object.entries(copy.slots))
    if (slot) stats.push({ key, value: slot.value, unit: null, label: slot.label, sub: slot.sub ?? null });
  return stats;
}

export const staleNote = (c: Contributions, copy: MetricsCopy) => `${copy.asOf} ${shortDate(c.range.to)}`;
```

```ts
"use client";

import { useCallback, useEffect, useRef, useState, type RefObject } from "react";
import { useReducedMotion } from "framer-motion";
import { gsap, ScrollTrigger } from "@/lib/gsap";
import { createSpeedometer, gateOpen } from "@/lib/metrics/seenGate";
import type { MorphRule } from "@/lib/metrics/settings";
import type { View } from "@/lib/metrics/types";

// Scroll decides flat or skyline, on one ScrollTrigger. The block's top
// crossing the line downward asks for the skyline; crossing it upward clears
// to flat. The skyline must be seen: it plays only once the chart is in view
// and the scroll has settled, or the scroll is slow enough to keep it in view
// for the whole morph; otherwise the block stays flat and pending, and plays
// the next time that holds (on a stop, or a slow return). Once played it
// persists anywhere below. A toggle click plays the same morph and holds until
// the next crossing. Reduced motion: no wait, and the engine snaps.
export function useScrollMorph(blockRef: RefObject<HTMLElement | null>, rule: MorphRule) {
  const [view, setView] = useState<View>("flat");
  const viewRef = useRef<View>("flat");
  const pendingRef = useRef(false);
  const reduced = useReducedMotion() ?? false;
  const show = useCallback((v: View) => {
    viewRef.current = v;
    setView(v);
  }, []);

  useEffect(() => {
    const block = blockRef.current;
    if (!block) return;
    const meter = createSpeedometer();
    let settleTimer = 0;
    let plays = 0;
    const hold = (on: boolean) => {
      pendingRef.current = on;
      block.dataset.morphPending = on ? "true" : "false";
    };
    const play = () => {
      if (viewRef.current !== "skyline") block.dataset.morphPlays = String(++plays);
      show("skyline");
    };
    const evaluate = () => {
      if (!pendingRef.current) return;
      const now = performance.now();
      const m = meter.read(now, rule.settledMs);
      const box = (block.querySelector<HTMLElement>("[data-skyline-stage]") ?? block).getBoundingClientRect();
      const open = gateOpen({
        top: box.top, height: box.height, viewport: window.innerHeight, share: rule.inViewShare,
        speed: m.speed, dir: m.dir, settled: m.settled, durationMs: rule.durationMs,
      });
      if (open) {
        hold(false);
        play();
      } else if (!m.settled) {
        window.clearTimeout(settleTimer);
        settleTimer = window.setTimeout(evaluate, Math.max(16, m.lastFastAt + rule.settledMs - now + 8));
      }
    };
    const onScroll = () => {
      meter.push(performance.now(), window.scrollY, rule.settleSpeed);
      evaluate();
    };
    const cross = (past: boolean) => {
      if (!past) {
        hold(false);
        show("flat");
      } else if (reduced) {
        play();
      } else {
        hold(true);
        evaluate();
      }
    };
    window.addEventListener("scroll", onScroll, { passive: true });
    window.addEventListener("resize", evaluate);
    const ctx = gsap.context(() => {
      // The range runs to the end of the page, so a fling past the whole block
      // still reads as one crossing (onEnter), never a skipped one.
      const st = ScrollTrigger.create({
        trigger: block,
        start: `top ${rule.triggerPct}%`,
        end: "max",
        invalidateOnRefresh: true,
        onEnter: () => cross(true),
        onLeaveBack: () => cross(false),
      });
      cross(st.scroll() > st.start);
    });
    // Anything above that reflows without a resize moves the line; re-measure
    // only when the block's own top moves (the morph growing the chart must not).
    let timer = 0;
    const blockTop = () => Math.round(block.getBoundingClientRect().top + window.scrollY);
    let lastTop = blockTop();
    const observer = new ResizeObserver(() => {
      const top = blockTop();
      if (top === lastTop) return;
      lastTop = top;
      window.clearTimeout(timer);
      timer = window.setTimeout(() => ScrollTrigger.refresh(), 150);
    });
    observer.observe(document.body);
    return () => {
      observer.disconnect();
      window.clearTimeout(timer);
      window.clearTimeout(settleTimer);
      window.removeEventListener("scroll", onScroll);
      window.removeEventListener("resize", evaluate);
      ctx.revert();
      pendingRef.current = false;
    };
  }, [blockRef, rule, reduced, show]);

  const choose = useCallback(
    (v: View) => {
      pendingRef.current = false;
      if (blockRef.current) blockRef.current.dataset.morphPending = "false";
      show(v);
    },
    [blockRef, show],
  );

  return { view, choose };
}
```
